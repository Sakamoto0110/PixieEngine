cmake_minimum_required(VERSION 3.25)

# Standalone installed consumer. Configure/refusal evidence alone never passes G1.
function(consumer_refuse reason)
    message(FATAL_ERROR "PIXIE_CONSUMER_REFUSED: ${reason}")
endfunction()

function(consumer_json output document)
    string(JSON value ERROR_VARIABLE error GET "${document}" ${ARGN})
    if(error)
        consumer_refuse("invalid manifest field ${ARGN}: ${error}")
    endif()
    set(${output} "${value}" PARENT_SCOPE)
endfunction()

function(consumer_digest value description)
    string(LENGTH "${value}" length)
    if(NOT length EQUAL 64 OR NOT "${value}" MATCHES "^[0-9a-f]+$")
        consumer_refuse("invalid ${description} SHA256")
    endif()
endfunction()

function(consumer_relative value)
    cmake_path(IS_ABSOLUTE value absolute)
    cmake_path(NORMAL_PATH value OUTPUT_VARIABLE normalized)
    if(absolute OR NOT "${value}" STREQUAL "${normalized}" OR "${value}" STREQUAL ""
       OR "${value}" MATCHES "(^|/)\\.\\.?(/|$)" OR "${value}" MATCHES "[:;\\\\\n\r]")
        consumer_refuse("unsafe SDK-relative payload path: ${value}")
    endif()
endfunction()

function(consumer_no_source_targets directory)
    get_property(targets DIRECTORY "${directory}" PROPERTY BUILDSYSTEM_TARGETS)
    if(targets)
        consumer_refuse("package created build targets instead of installed imports: ${targets}")
    endif()
    get_property(children DIRECTORY "${directory}" PROPERTY SUBDIRECTORIES)
    foreach(child IN LISTS children)
        consumer_no_source_targets("${child}")
    endforeach()
endfunction()
