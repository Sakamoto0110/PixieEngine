cmake_minimum_required(VERSION 3.25)
project(PixieSharedBoundaryGate LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PIXIE_BUILD_SAMPLES OFF CACHE BOOL "" FORCE)
set(PIXIE_ENGINE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
get_filename_component(ENGINE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_subdirectory("${ENGINE}" engine)

# A standalone experiment: production targets and their runtime choice are unchanged.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
function(shared_runtime directory)
    get_property(targets DIRECTORY "${directory}" PROPERTY BUILDSYSTEM_TARGETS)
    foreach(target IN LISTS targets)
        set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "${CMAKE_MSVC_RUNTIME_LIBRARY}")
    endforeach()
    get_property(children DIRECTORY "${directory}" PROPERTY SUBDIRECTORIES)
    foreach(child IN LISTS children)
        shared_runtime("${child}")
    endforeach()
endfunction()
shared_runtime("${ENGINE}")

set(sources)
foreach(target pixie_core pixie_platform pixie_render pixie_app pixie_runtime)
    get_target_property(files ${target} SOURCES)
    get_target_property(directory ${target} SOURCE_DIR)
    foreach(file IN LISTS files)
        list(APPEND sources "${directory}/${file}")
    endforeach()
endforeach()
add_library(boundary-engine SHARED ${sources} "${ENGINE}/capi/src/Capi.cpp" GateEngine.cpp)
target_compile_options(boundary-engine PRIVATE /MP /W4 /WX /permissive- /utf-8)
set_target_properties(boundary-engine PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_compile_definitions(boundary-engine PRIVATE GLEW_STATIC PIXIE_CAPI_BUILD GLFW_INCLUDE_NONE PIXIE_CAPI_FACTORY
    PIXIE_CORE_BUILD PIXIE_PLATFORM_BUILD PIXIE_RENDER_BUILD PIXIE_APP_BUILD PIXIE_RUNTIME_BUILD)
target_link_libraries(boundary-engine PUBLIC pixie::pixie pixie::math EnTT::EnTT glew
    PRIVATE glfw stb_truetype nlohmann_json::nlohmann_json box2d shell32 ole32)
foreach(part core platform render app)
    target_include_directories(boundary-engine PUBLIC "${ENGINE}/engine/${part}/include")
endforeach()
target_include_directories(boundary-engine PUBLIC "${ENGINE}/runtime/include" "${ENGINE}/capi/include"
    PRIVATE "${ENGINE}/external/stb")

add_library(boundary-module MODULE GateModule.cpp)
option(BOUNDARY_BREAK_IDENTITY "Deliberately change the module type identity for sensitivity testing" OFF)
if(BOUNDARY_BREAK_IDENTITY)
    target_compile_definitions(boundary-module PRIVATE BOUNDARY_BREAK_IDENTITY)
endif()
target_link_libraries(boundary-module PRIVATE boundary-engine)
add_executable(boundary-host GateHost.cpp)
target_link_libraries(boundary-host PRIVATE boundary-engine glfw)
add_custom_command(TARGET boundary-host POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    $<TARGET_FILE:glfw> $<TARGET_FILE_DIR:boundary-host>)
enable_testing()
add_test(NAME shared-boundary COMMAND boundary-host $<TARGET_FILE:boundary-module>)
