cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    file(READ "${CMAKE_CURRENT_LIST_DIR}/../../version.json" version_manifest)
    string(JSON export_version GET "${version_manifest}" version)
    project(PixieExporter VERSION "${export_version}" LANGUAGES CXX)
    if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "The paired exporter requires Windows x64 MSVC")
    endif()
    if(NOT IS_ABSOLUTE "${PIXIE_SDK_ROOT}" OR NOT PIXIE_SDK_PROFILE STREQUAL "shipping")
        message(FATAL_ERROR "Select an explicit absolute public SDK root and PIXIE_SDK_PROFILE=shipping")
    endif()
    find_package(PixieEngine CONFIG REQUIRED PATHS "${PIXIE_SDK_ROOT}/lib/cmake/PixieEngine" NO_DEFAULT_PATH)
    if(NOT IS_ABSOLUTE "${PIXIE_EXPORT_JSON_INCLUDE_DIR}" OR
       NOT EXISTS "${PIXIE_EXPORT_JSON_INCLUDE_DIR}/nlohmann/json.hpp")
        message(FATAL_ERROR "Supply PIXIE_EXPORT_JSON_INCLUDE_DIR for the exporter's private JSON header dependency")
    endif()
    add_library(pixie_export_json INTERFACE)
    target_include_directories(pixie_export_json SYSTEM INTERFACE "${PIXIE_EXPORT_JSON_INCLUDE_DIR}")
    add_library(nlohmann_json::nlohmann_json ALIAS pixie_export_json)
    include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/PixieTargetDefaults.cmake")
    set(PixieEngine_VERSION "${export_version}")
endif()

add_executable(pixie-export main.cpp Export.cpp PeImage.cpp Process.cpp Sdk.cpp Product.cpp)
target_link_libraries(pixie-export PRIVATE Pixie::Runtime nlohmann_json::nlohmann_json)
if(WIN32)
    target_link_libraries(pixie-export PRIVATE bcrypt)
endif()
target_compile_definitions(pixie-export PRIVATE
    "PIXIE_EXPORT_TOOL_VERSION=\"${PixieEngine_VERSION}\"")
pixie_target_defaults(pixie-export)
set_target_properties(pixie-export PROPERTIES FOLDER tools)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    pixie_stage_runtime(pixie-export)
endif()
